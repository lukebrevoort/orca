Use the existing named headless agent-browser session, real scratch login and the steps above. Do not force the theme click. Run this read-only expression with agent-browser --session bre387-built-623e5fc eval; the recorded baseline fails both the hit target and non-overlap conditions.

(() => {
  const header = document.querySelector('.desktop-workspace-header');
  const toolbar = document.querySelector('.bulk-action-bar');
  const theme = document.querySelector('.desktop-theme-toggle');
  const search = document.querySelector('.desktop-global-search input');
  const hit = (element) => {
    const r = element.getBoundingClientRect();
    const target = document.elementFromPoint(r.x + r.width/2, r.y + r.height/2);
    return target === element || element.contains(target);
  };
  const h = header.getBoundingClientRect(), b = toolbar.getBoundingClientRect();
  return { pass: b.top >= h.bottom && hit(theme) && hit(search),
    headerBottom: h.bottom, toolbarTop: b.top,
    themeHit: hit(theme), searchHit: hit(search),
    selection: toolbar.innerText };
})()
